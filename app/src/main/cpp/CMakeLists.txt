cmake_minimum_required(VERSION 3.22.1)
project(outpost_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.local/llama.cpp" ABSOLUTE)
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Run scripts/prepare-native.py before building")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
# Baseline x86_64, including emulators without AVX feature passthrough.
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
set(GGML_SSE42 OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(outpost_engine SHARED engine.cpp attention_probe.cpp speculation.cpp q2_kernel.c q2_batch.c q2_arm.c cpu_caps.c q2_dispatch.c)
target_include_directories(outpost_engine PRIVATE "${LLAMA_SOURCE}/ggml/src" "${LLAMA_SOURCE}/ggml/src/ggml-cpu")
set_source_files_properties(q2_kernel.c q2_batch.c q2_arm.c PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
target_link_options(outpost_engine PRIVATE "-Wl,--wrap=ggml_vec_dot_q2_0_q8_0")
target_link_options(outpost_engine PRIVATE "-Wl,--wrap=ggml_compute_forward_mul_mat_tiled")
target_link_options(outpost_engine PRIVATE "-Wl,--wrap=ggml_compute_forward_flash_attn_ext")
option(OUTPOST_RESEARCH "Expose research JNI entry points" OFF)
if(OUTPOST_RESEARCH)
    target_compile_definitions(outpost_engine PRIVATE OUTPOST_RESEARCH=1)
else()
    target_compile_definitions(outpost_engine PRIVATE OUTPOST_RESEARCH=0)
    target_link_options(outpost_engine PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/release.exports")
    set_property(TARGET outpost_engine APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/release.exports")
endif()
target_compile_options(outpost_engine PRIVATE -O3)
target_link_libraries(outpost_engine PRIVATE llama android log)
